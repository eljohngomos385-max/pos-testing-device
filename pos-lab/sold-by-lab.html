5 · Code

Each item already saves soldBy ('each' or 'measure') and a unit. So there's no new data and no migration: the change only cuts the list and adds the pad.

Now · pos-items.js
// 13 units, and a function to guess soldBy from the unit const ITEM_UNITS = [['pc','box','bag','set','pair','roll', 'sheet','can'], ['m','ft','kg','L','gal']]; const soldByFor = (unit, was) => ITEM_UNITS[1].includes(unit) ? 'measure' : ITEM_UNITS[0].includes(unit) ? 'each' : was || 'each'; ifr('Sold per', ipick('unit', E.unit)) // picking a unit sets soldBy behind the owner's back E.soldBy = soldByFor(u, E.soldBy); E.unit = u;
Now · pos-sell.js addToCart
// a weighed item adds 1, like a hammer it.qty += 1;
After · pos-items.js
// the owner picks soldBy; unit only matters for Weight const PER_UNITS = ['kg', 'g', 'lb', 'm', 'L']; ifr('Sold by', ipick('soldBy', E.soldBy === 'measure' ? 'Weight' : 'Piece')) E.soldBy === 'measure' && ifr('Per', ipick('unit', E.unit)) // Piece → unit 'pc'; Weight → unit 'kg' to start
After · pos-sell.js addToCart
// one line: a weighed item asks first (the price-ask pad, reused) if (!line && p.soldBy === 'measure') return openEditSheet('weight', p.id, via); it.qty += 1;
