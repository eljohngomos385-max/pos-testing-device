5 · Code

Each item already saves soldBy + unit, so there's no new data. The page sets soldBy itself, so soldByFor (guessing it from the unit) goes.

Now · pos-items.js
const ITEM_UNITS = [['pc','box','bag','set','pair','roll', 'sheet','can'], ['m','ft','kg','L','gal']]; const soldByFor = (unit, was) => ITEM_UNITS[1].includes(unit) ? 'measure' : ITEM_UNITS[0].includes(unit) ? 'each' : was || 'each'; ifr('Sold per', ipick('unit', E.unit))
Now · pos-sell.js addToCart
// a measured item adds 1, like a hammer it.qty += 1;
After · pos-items.js
const MEASURES = ['kg', '100 g', 'L', 'mL', 'm', 'm²']; // a row with ›, like Categories; its page reuses itemTickPage inav('Sold by', E.soldBy === 'measure' ? `Measure · ${E.unit}` : 'Piece', 'soldby') // page: Piece / Measure, then Unit (MEASURES) under Measure // Price reads "₱65.00 / kg" when soldBy is 'measure'
After · pos-sell.js addToCart
// one line: measured asks first (the price-ask pad, reused) if (!line && p.soldBy === 'measure') return openEditSheet('amount', p.id, via); it.qty += 1;
